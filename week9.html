@Transactional
public WorkoutCompleteResponse completeWorkout(
        Long workoutId,
        Long currentUserId,
        WorkoutCompleteRequest request
) {
    // 1. 운동을 조회하면서 해당 행을 잠금
    Workout workout = workoutRepository
            .findByIdForUpdate(workoutId)
            .orElseThrow(() ->
                    new NotFoundException("운동 기록이 없습니다."));

    // 2. 현재 사용자의 운동인지 확인
    checkOwner(workout, currentUserId);

    // 3. 이미 완료된 운동인지 등 상태 확인
    validateCompletable(workout);

    RunningDetails details = null;

    if (workout.isRunning()) {
        // 4. 러닝이면 distanceM 필수
        BigDecimal distance = validateDistance(request);

        // 5. 러닝 상세 정보 조회
        details = runningDetailsRepository
                .findById(workoutId)
                .orElseThrow(() ->
                        new NotFoundException("러닝 상세가 없습니다."));

        // 6. 최종 거리 저장
        details.updateDistance(distance);
    }

    // 7. status / endedAt / durationSeconds 변경
    workout.complete(LocalDateTime.now(workoutClock));
     //RDP수행으로 밀리는 시각을 방지하여 complete먼저 실행
    if (details != null) {
        // 8. 저장된 GPS 경로를 RDP 2.5m로 단순화
        simplifyRoute(workoutId);
    }

    // 9. 완료된 운동 정보를 Response DTO로 변환
    return WorkoutCompleteResponse.from(workout);
}

④ DB에서 운동 데이터를 조회하는 부분

WorkoutRepository.java
public interface WorkoutRepository
        extends JpaRepository<Workout, Long> {

    // 같은 운동을 동시에 완료하려는 상황 방지
    @Lock(LockModeType.PESSIMISTIC_WRITE)

    // workoutId에 해당하는 운동을 조회
    @Query("""
        select w
        from Workout w
        where w.workoutId = :workoutId
    """)
    Optional<Workout> findByIdForUpdate(
            @Param("workoutId") Long workoutId
    );
}

PESSIMISTIC_WRITE

조회한 운동 행에 DB Lock을 걸어 다른 트랜잭션이 동시에 수정하지 못하도록 함

  • 중복 완료 방지
  • 동시에 상태가 변경되는 문제 방지
  • 한 트랜잭션이 끝난 뒤 다음 수정 허용
08 · Entity
Workout.java

⑤ 실제 운동 상태를 바꾸는 부분

Workout.java
@Entity
@Table(name = "workouts")
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class Workout {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "workout_id")
    private Long workoutId;

    @Column(name = "user_id", nullable = false)
    private Long userId;

    @Column(name = "workout_type", nullable = false, length = 16)
    private String workoutType;

    @Column(nullable = false, length = 16)
    private String status;

    @Column(name = "started_at", nullable = false)
    private LocalDateTime startedAt;

    @Column(name = "ended_at")
    private LocalDateTime endedAt;

    // MySQL INT UNSIGNED의 전체 범위를 담기 위해 Long 사용
    @Column(name = "duration_seconds", nullable = false)
    private Long durationSeconds;

    @Column(name = "paused_at")
    private LocalDateTime pausedAt;

    @Column(name = "paused_seconds", nullable = false)
    private Long pausedSeconds;

    public boolean isCompleted() {
        return "COMPLETED".equals(status);
    }

    public boolean isRunning() {
        return "RUNNING".equals(workoutType);
    }

    public boolean canComplete() {
        return "IN_PROGRESS".equals(status) || "PAUSED".equals(status);
    }

    public void complete(LocalDateTime serverNow) {
        // 1. 완료 가능한 상태인지 확인
        if (!canComplete()) {
            throw new IllegalStateException("완료할 수 없는 운동 상태입니다.");
        }

        // 2. 시작 시각과 누적 일시정지 시간 검증
        if (startedAt == null || serverNow.isBefore(startedAt)) {
            throw new IllegalStateException("운동 시작 시각이 올바르지 않습니다.");
        }

        if (pausedSeconds == null || pausedSeconds < 0) {
            throw new IllegalStateException("누적 일시정지 시간이 올바르지 않습니다.");
        }

        long totalPausedSeconds = pausedSeconds;

        // 3. 일시정지 중 완료했다면 마지막 정지 시간도 합산
        if ("PAUSED".equals(status)) {
            if (pausedAt == null || serverNow.isBefore(pausedAt)) {
                throw new IllegalStateException("일시정지 시각이 올바르지 않습니다.");
            }

            totalPausedSeconds += Duration.between(
                    pausedAt, serverNow
            ).getSeconds();
        } else if (pausedAt != null) {
            throw new IllegalStateException("운동 상태와 일시정지 시각이 일치하지 않습니다.");
        }

        // 4. 전체 경과 시간에서 일시정지 시간을 제외
        long elapsedSeconds = Duration.between(
                startedAt, serverNow
        ).getSeconds();

        long activeSeconds = elapsedSeconds - totalPausedSeconds;

        // 5. 음수 여부와 DB에 저장 가능한 범위 확인
        if (activeSeconds < 0
                || activeSeconds > 4_294_967_295L
                || totalPausedSeconds > 4_294_967_295L) {
            throw new IllegalStateException("운동 시간 계산 결과가 올바르지 않습니다.");
        }

        // 6. 종료 시각·순수 운동 시간을 저장하고 완료 상태로 변경
        this.endedAt = serverNow;
        this.pausedSeconds = totalPausedSeconds;
        this.pausedAt = null;
        this.durationSeconds = activeSeconds;
        this.status = "COMPLETED";
    }
}

Entity에서 바뀌는 값

  • status → COMPLETED
  • endedAt → 현재 종료 시간
  • durationSeconds → 실제 운동 시간
Entity는 트랜잭션 안에서 이미 조회된 상태이므로 값만 바꿔도 JPA가 변경을 감지함.

public RouteSimplifier(
    // RDP 적용 여부 (기본값: true)
    @Value("${app.running.rdp.enabled:true}")
    boolean enabled,

    // RDP epsilon 값 (기본값: 2.5m)
    @Value("${app.running.rdp.epsilon-m:2.5}")
    double epsilonM
) {
    // epsilon은 0보다 큰 정상적인 숫자만 허용
    if (!Double.isFinite(epsilonM) || epsilonM <= 0) {
        throw new IllegalArgumentException(
            "RDP epsilon은 0보다 큰 유한한 값이어야 합니다."
        );
    }

    this.enabled = enabled;
    this.epsilonM = epsilonM;
}

RDP구현중 수정해야하는 값

epsilon엡실론 = 두 유지점 사이 직선에서 중간 좌표가 벗어날 수 있는 허용 거리 기본값은 2m, 값에 따라 좌표수가 달라짐

14 · 결론 / 다음 주
9주차 마무리

현재 RDP epsilon 후보: 2.5m

2.5m를 선택한 이유

  • 1m: 형태는 잘 유지되지만 좌표 감소 효과가 상대적으로 낮음
  • 2m: 안정적이지만 2.5m보다 좌표가 더 많이 남음
  • 2.5m: 2m와 형태 차이는 거의 없고 좌표는 더 감소
  • 3m 이상: 곡선이 많은 코스에서 단순화가 보이기 시작
  • 5m: 트랙 곡선 형태 변화가 큼

실제 ShareFit 사용자 GPS 데이터가 확보되면 다시 조정할 수 있음
